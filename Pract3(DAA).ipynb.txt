        "arr = []\n",

        "\n",
        "# Measure Execution Time\n",
        "start = time.perf_counter()\n",
        "\n",
        "# Convert elements to negative for Max Heap\n",
        "max_heap = [-x for x in arr]\n",
        "heapq.heapify(max_heap)\n",
